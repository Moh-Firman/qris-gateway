Dokumentasi API
Integrasikan gateway ke sistem apapun via REST API.
Base URL

Autentikasi

Semua endpoint internal membutuhkan secret yang diset di Pengaturan. Sertakan sebagai field secret di body JSON.

1. Buat Pembayaran POST /internal_create
import requests

BASE = "https://apiv1.mxthz.xyz/api"
SECRET = "SECRET_KAMU"

# 1. Buat pembayaran
r = requests.post(BASE + "/internal_create", json={
    "secret": SECRET,
    "telegram_id": 123456789,
    "amount": 50000
}).json()
print(r["qris_string"])  # render jadi QR
print(r["total"])        # nominal dibayar

# 2. Cek status
s = requests.post(BASE + "/internal_status",
    json={"secret": SECRET, "id": r["id"]}).json()
print(s["status"])  # pending | paid | expired

# 3. Approve (saldo diurus sistem kamu)
requests.post(BASE + "/internal_approve",
    json={"secret": SECRET, "id": r["id"], "no_credit": True})
# 1. Buat pembayaran
curl -X POST https://apiv1.mxthz.xyz/api/internal_create \
  -H "Content-Type: application/json" \
  -d '{"secret":"SECRET_KAMU","telegram_id":123456789,"amount":50000}'

# 2. Cek status

  -H "Content-Type: application/json" \

<?php
$base = "https://apiv1.mxthz.xyz/api";
$secret = "SECRET_KAMU";

function gw($ep, $data) {
    global $base;
    $ch = curl_init($base . "/" . $ep);
    curl_setopt_array($ch, [
        CURLOPT_POST => true,
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_HTTPHEADER => ["Content-Type: application/json"],
        CURLOPT_POSTFIELDS => json_encode($data)
    ]);
    return json_decode(curl_exec($ch), true);
}

// 1. Buat pembayaran
$r = gw("internal_create", [
    "secret" => $secret,
    "telegram_id" => 123456789,
    "amount" => 50000
]);
echo $r["qris_string"]; // render jadi QR

// 2. Cek status
$s = gw("internal_status",
    ["secret" => $secret, "id" => $r["id"]]);
echo $s["status"];

// 3. Approve
gw("internal_approve", [
    "secret" => $secret, "id" => $r["id"],
    "no_credit" => true
]);
const BASE = "https://apiv1.mxthz.xyz/api";
const SECRET = "SECRET_KAMU";

async function gw(ep, data) {
  const r = await fetch(BASE + "/" + ep, {
    method: "POST",
    headers: {"Content-Type": "application/json"},
    body: JSON.stringify({secret: SECRET, ...data})
  });
  return r.json();
}

// 1. Buat pembayaran
const pay = await gw("internal_create",
  {telegram_id: 123456789, amount: 50000});
console.log(pay.qris_string); // render jadi QR

// 2. Cek status
const st = await gw("internal_status", {id: pay.id});
console.log(st.status); // pending | paid | expired

// 3. Approve
await gw("internal_approve",
  {id: pay.id, no_credit: true});
